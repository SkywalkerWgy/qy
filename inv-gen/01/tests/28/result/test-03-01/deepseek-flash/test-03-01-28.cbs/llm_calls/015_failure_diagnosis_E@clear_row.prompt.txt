/*@

  logic integer abs_int(integer x) = x < 0 ? -x : x;

  predicate valid_board(integer n, int *board) =
    1 <= n <= 46340 &&
    \valid(board + (0 .. n * n - 1));

  predicate binary_board{L}(integer n, int *board) =
    \forall integer r, c; 0 <= r < n && 0 <= c < n ==>
      \at(board[idx(n, r, c)], L) == 0 ||
      \at(board[idx(n, r, c)], L) == 1;

  predicate zero_board{L}(integer n, int *board) =
    \forall integer r, c; 0 <= r < n && 0 <= c < n ==>
      \at(board[idx(n, r, c)], L) == 0;

  predicate one_queen_in_row{L}(integer n, int *board, integer row) =
    0 <= row < n &&
    \exists integer col;
      0 <= col < n &&
      \at(board[idx(n, row, col)], L) == 1 &&
      (\forall integer j; 0 <= j < n && j != col ==>
        \at(board[idx(n, row, j)], L) == 0);

  predicate rows_solved{L}(integer n, int *board, integer row_limit) =
    0 <= row_limit <= n &&
    \forall integer r; 0 <= r < row_limit ==>
      one_queen_in_row{L}(n, board, r);

    0 <= row_start <= n &&
    \forall integer r, c; row_start <= r < n && 0 <= c < n ==>

    0 <= row_limit <= n &&
    \forall integer r1, c1, r2, c2;

      \at(board[idx(n, r2, c2)], L) == 1 ==>

    rows_solved{L}(n, board, n) &&

    \forall int *board, integer n, row_limit;

      rows_solved{L}(n, board, row_limit) &&
      one_queen_in_row{L}(n, board, row_limit)

  requires valid_board(n, board);
  assigns board[0 .. n * n - 1];
  ensures zero_board{Here}(n, board);

_Safe void clear_board(int n, int board[_Nonnull ])
{

      // Loop A
/*@
      loop invariant i_0: valid_board(n, board);

      loop invariant i_1: 0 <= i <= n * n;

      loop invariant i_2: \forall integer k; 0 <= k < i ==> board[k] == 0;

        board[i] = 0;
        i++;
    }

_Safe int is_safe(int n, int board[_Nonnull ], int row, int col)
{
    int r = 0;

      // Loop B
/*@
      loop invariant i_3: 0 <= r <= row;

      loop invariant i_4: valid_board(n, board);

      loop invariant i_5: \forall integer r1, c1; 0 <= r1 < r && 0 <= c1 < n && board[idx(n, r1, c1)] == 1 ==> c1 != col && abs_int(row - r1) != abs_int(col - c1);


      loop assigns r;
      loop variant row - r;
    */
    while (r < row) {
        int c = 0;

              // Loop C
/*@
          loop invariant i_6: 0 <= c <= n;

          loop invariant i_7: valid_board(n, board);

          loop invariant i_8: \forall integer c1; 0 <= c1 < c && board[idx(n, r, c1)] == 1 ==> c1 != col && abs_int(row - r) != abs_int(col - c1);


          loop assigns c;
          loop variant n - c;
        */
        while (c < n) {
            if (board[r * n + c] == 1) {
                if (c == col) {
                    return 0;
                }
                if (row - r == col - c || row - r == c - col) {
                    return 0;
                }
            }
            c++;
        }
        r++;
    }

    return 1;
}

_Safe int solve_from_row(int n, int board[_Nonnull ], int row)
{
    if (row == n) {
        return 1;
    }

    int col = 0;

      // Loop F
/*@
      loop invariant i_28: valid_board(n, board);

      loop invariant i_29: 0 <= row < n;

      loop invariant i_30: 0 <= col <= n;

      loop invariant i_31: binary_board(n, board);

      loop invariant i_34: rows_empty_from(n, board, row);


      loop assigns col, board[row * n .. n * n - 1];

    while (col < n) {
        if (is_safe(n, board, row, col)) {
            //@ assert row < n;

            place_queen(n, board, row, col);
            //@ assert (row + 1) * n <= n * n;
            if (solve_from_row(n, board, row + 1)) {
                return 1;

            clear_row(n, board, row);
        }

    }

    return 0;
}
